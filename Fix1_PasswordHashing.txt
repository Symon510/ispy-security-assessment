 Strategy:  Add PBKDF2-based PasswordHasher class. Login now hashes the
            attempted password with the stored salt and compares to the
            stored hash, instead of decrypting the stored password.

NEW FILE: PasswordHasher.cs

using System;
using System.Security.Cryptography;
using System.Text;

namespace iSpyApplication
{
    /// <summary>
    /// One-way password hashing using PBKDF2-HMAC-SHA256.
    /// Replaces the previous use of EncDec.EncryptData/DecryptData for
    /// password storage (which was a reversible AES round-trip with a
    /// hardcoded salt -- see CWE-257, CWE-760).
    /// </summary>
    public static class PasswordHasher
    {
        private const int SaltSize    = 16;     // 128-bit random salt
        private const int HashSize    = 32;     // 256-bit derived key
        private const int Iterations  = 100000; // OWASP 2023 minimum for PBKDF2-SHA256
        private const string Prefix   = "PBKDF2$";

        /// <summary>
        /// Returns "PBKDF2$<iter>$<base64Salt>$<base64Hash>" for a plaintext password.
        /// </summary>
        public static string Hash(string password)
        {
            if (password == null) throw new ArgumentNullException("password");

            var salt = new byte[SaltSize];
            using (var rng = RandomNumberGenerator.Create())
                rng.GetBytes(salt);

            byte[] hash;
            using (var pbkdf2 = new Rfc2898DeriveBytes(password, salt, Iterations,
                                                      HashAlgorithmName.SHA256))
            {
                hash = pbkdf2.GetBytes(HashSize);
            }

            return Prefix + Iterations + "$"
                          + Convert.ToBase64String(salt) + "$"
                          + Convert.ToBase64String(hash);
        }

        /// <summary>
        /// Constant-time verification of a plaintext password against a stored hash.
        /// </summary>
        public static bool Verify(string password, string stored)
        {
            if (password == null || stored == null) return false;
            if (!stored.StartsWith(Prefix)) return false;

            var parts = stored.Split('$');
            if (parts.Length != 4) return false;

            int iter;
            if (!int.TryParse(parts[1], out iter)) return false;

            byte[] salt, expected;
            try
            {
                salt     = Convert.FromBase64String(parts[2]);
                expected = Convert.FromBase64String(parts[3]);
            }
            catch (FormatException) { return false; }

            byte[] actual;
            using (var pbkdf2 = new Rfc2898DeriveBytes(password, salt, iter,
                                                      HashAlgorithmName.SHA256))
            {
                actual = pbkdf2.GetBytes(expected.Length);
            }

            // Constant-time comparison to prevent timing attacks
            int diff = expected.Length ^ actual.Length;
            for (int i = 0; i < expected.Length && i < actual.Length; i++)
                diff |= expected[i] ^ actual[i];
            return diff == 0;
        }

        /// <summary>True if the stored value uses the new PBKDF2 format.</summary>
        public static bool IsHashed(string stored)
        {
            return stored != null && stored.StartsWith(Prefix);
        }
    }
}

 RESULT:

 - Stored credential is no longer recoverable to plaintext, even if both the
   config file AND the source code are exposed.
 - Salt is now per-record and 128 bits of crypto-RNG output, replacing the
   hardcoded literal "Salt".
 - 100,000 PBKDF2-SHA256 iterations meet OWASP 2023 minimum guidance.
 - Constant-time comparison eliminates the timing side channel.
 - Legacy records are migrated transparently on next successful login;
   no forced password reset is required.
